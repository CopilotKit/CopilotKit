## Overview

`FrontendTool` describes a tool that runs in the browser and can be invoked by an agent. You register frontend tools via [`CopilotKitCoreConfig.tools`](/reference/core/types/CopilotKitCoreConfig) (or at runtime on [`CopilotKitCore`](/reference/core/classes/CopilotKitCore)). A tool has a name, an optional parameter schema, and a handler that receives the parsed arguments and an execution context.

## Import

```ts
import type { FrontendTool } from "@copilotkit/core";
```

## Definition

`FrontendTool<T>` takes the argument type `T` (default `Record<string, unknown>`). This tool sets every public field, and its handler names every field of the `FrontendToolHandlerContext` it receives:

```ts
import type { FrontendTool } from "@copilotkit/core";
import { z } from "zod";

const lookupOrder = {
  name: "lookupOrder",
  description: "Look up an order by its id",
  parameters: z.object({ orderId: z.string() }),
  handler: async ({ orderId }, { toolCall, agent, signal, isReplay }) => {
    return `Order ${orderId} is on its way`;
  },
  followUp: true,
  reconnectBehavior: "passive",
  agentId: "default",
  available: true,
  webmcp: { annotations: { readOnlyHint: true } },
} satisfies FrontendTool<{ orderId: string }>;
```

## Properties

<PropertyReference name="name" type="string" required>
  Unique name of the tool. This is the name the agent uses to call it.
</PropertyReference>

<PropertyReference name="description" type="string">
  Human-readable description of what the tool does. Helps the model decide when to call it.
</PropertyReference>

<PropertyReference name="parameters" type="StandardSchemaV1<any, T>">
  Standard Schema describing the tool's arguments. It defines the advertised
  tool schema and types the `args` passed to `handler`. Runtime arguments are
  JSON-parsed but are not validated against that schema.
</PropertyReference>

<PropertyReference name="handler" type="(args: T, context: FrontendToolHandlerContext) => Promise<unknown>">
  Function executed when the agent calls the tool. Receives the parsed `args` and an execution context.

  <PropertyReference name="context.toolCall" type="ToolCall">
    The AG-UI tool call that triggered this handler.
  </PropertyReference>

  <PropertyReference name="context.agent" type="AbstractAgent | undefined">
    The agent instance that issued the tool call. Absent when a browser agent calls the tool through WebMCP.
  </PropertyReference>

  <PropertyReference name="context.signal" type="AbortSignal">
    Aborted when `stopAgent()` is called. Handlers can check `signal.aborted` or pass the signal to `fetch`/`setTimeout` to cooperatively cancel.
  </PropertyReference>

  <PropertyReference name="context.isReplay" type="boolean">
    `true` when the call is run while the thread's history is restored (see `reconnectBehavior`). Absent for a call from a live run.
  </PropertyReference>
</PropertyReference>

<PropertyReference name="followUp" type="boolean">
  Whether the agent should run again after the tool result is returned.
</PropertyReference>

<PropertyReference name="reconnectBehavior" type='"passive" | "resume-pending"' default='"passive"'>
  What happens to a call of this tool that is still unanswered when the thread's history is restored, for example after a page reload. With `"passive"`, the call is shown but the handler does not run. With `"resume-pending"`, the handler runs for that call with `isReplay: true` in its context, and the run then continues as usual. Opt in only for a handler that is safe to run more than once. A call does not start twice at the same time on one page, but it can run again after a stop or a thread switch if its result never reached the agent, and two tabs that restore the same thread can each run it. If the agent calls this tool together with a tool that stays passive, the follow-up run sends the passive call without a result, which some model providers reject.
</PropertyReference>

<PropertyReference name="agentId" type="string">
  Optional agent ID to constrain this tool to a specific agent. If specified, the tool is only available to that agent.
</PropertyReference>

<PropertyReference name="available" type="boolean" default="true">
  Whether this tool is available to the agent. Set to `false` to hide the tool without unregistering it.
</PropertyReference>

<PropertyReference name="webmcp" type="boolean | WebMCPToolConfig">
  Also exposes the tool to browser agents through the WebMCP API (`document.modelContext`). Pass `true` for default annotations, or `{ annotations: { readOnlyHint, untrustedContentHint } }`. When a browser agent calls the tool, the handler context has no `agent`. No-op where WebMCP is not available.
</PropertyReference>

## Usage

```ts
import type { FrontendTool } from "@copilotkit/core";
import { z } from "zod";

const setThemeTool: FrontendTool<{ theme: "light" | "dark" }> = {
  name: "setTheme",
  description: "Switch the app between light and dark mode",
  parameters: z.object({ theme: z.enum(["light", "dark"]) }),
  handler: async ({ theme }, { signal }) => {
    if (signal?.aborted) return;
    document.documentElement.dataset.theme = theme;
    return `Theme set to ${theme}`;
  },
};
```
