Start the Google Antigravity agent server. It hosts one AG-UI endpoint per
demo agent, mounted by name:

```bash title="Agent application"
python -m uvicorn agent_server:app --app-dir src --host 0.0.0.0 --port 8000
```

If you chose the CLI starter instead, keep its `npm run dev` process running.
Point Channels at the agent you want to drive; the neutral default agent is
mounted at `/default`:

```dotenv title=".env"
AGENT_URL=http://localhost:8000/default
```

Install the HTTP adapter in the Channels process:

```bash title="Terminal"
npm install @ag-ui/client@0.0.57
```

Create a new HTTP adapter for every channel thread:

```ts title="agent.ts"
import { HttpAgent } from "@ag-ui/client";

export function makeAgent(threadId: string) {
  const agent = new HttpAgent({ url: process.env.AGENT_URL! });
  agent.threadId = threadId;
  return agent;
}
```
